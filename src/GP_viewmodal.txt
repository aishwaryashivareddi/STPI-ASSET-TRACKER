      {showViewModal && (
        <div className="modal">
          <div className="modal-content">
            <h2>📎 Attached Files — {showViewModal.gateway_pass_id}</h2>
            <div style={{ display: 'flex', flexDirection: 'column', gap: '16px', marginTop: '16px' }}>
              {showViewModal.signed_copy ? (
                <div style={{ border: '2px solid #e2e8f0', borderRadius: '10px', overflow: 'hidden' }}>
                  <div style={{ padding: '12px 18px', background: '#f7fafc', display: 'flex', alignItems: 'center', justifyContent: 'space-between', borderBottom: '1px solid #e2e8f0' }}>
                    <span style={{ fontWeight: 700, color: '#2d3748' }}>📄 Signed Copy</span>
                    <div style={{ display: 'flex', gap: '8px' }}>

                      <button onClick={() => handleDownloadSignedCopy(showViewModal)} style={{ padding: '6px 14px', background: '#38a169', color: 'white', borderRadius: '6px', border: 'none', fontSize: '12px', fontWeight: 600, cursor: 'pointer', width: 'auto' }}>Download</button>
                      <button onClick={() => { handleDeleteSignedCopy(showViewModal); setShowViewModal(null); }} style={{ padding: '6px 14px', background: '#e53e3e', color: 'white', borderRadius: '6px', border: 'none', fontSize: '12px', fontWeight: 600, cursor: 'pointer', width: 'auto' }}>Delete</button>
                    </div>
                  </div>
                  <div style={{ padding: '12px', background: 'white', maxHeight: '300px', overflow: 'auto' }}>
                    {/\.(jpg|jpeg|png|gif)$/i.test(showViewModal.signed_copy) ? (
                      <img src={getSignedCopyUrl(showViewModal.signed_copy)} alt="Signed Copy" style={{ maxWidth: '100%', borderRadius: '6px' }} />
                    ) : /\.pdf$/i.test(showViewModal.signed_copy) ? (
                      <iframe src={getSignedCopyUrl(showViewModal.signed_copy)} title="Signed Copy" style={{ width: '100%', height: '280px', border: 'none', borderRadius: '6px' }} />

                      <p style={{ color: '#718096', textAlign: 'center', padding: '20px' }}>Preview not available — use View button to open</p>
                    )}
                  </div>
                </div>
              ) : (
                <p style={{ textAlign: 'center', color: '#718096', padding: '30px' }}>No signed copy uploaded yet</p>
              )}
            </div>
            <div className="form-actions">
              <button type="button" onClick={() => setShowViewModal(null)}>Close</button>
            </div>
          </div>
        </div>
      )}
